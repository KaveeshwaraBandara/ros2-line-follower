# GPU (CUDA 12.1) variant of requirements.txt — for local training on an
# NVIDIA GPU (e.g. RTX 4080). Not used by CI or the Codespaces devcontainer,
# both of which have no GPU.
#
# Standalone file — install this INSTEAD OF requirements.txt, not alongside
# it (both pin "torch"; installing both in one pip invocation causes a
# double-requirement conflict):
#   pip3 install -r requirements-gpu.txt
#
# Why --extra-index-url instead of --index-url: this keeps PyPI as the
# primary index for numpy/opencv/etc. and only adds the CUDA wheel server
# as a secondary source. --index-url would replace PyPI entirely and break
# every other package install.
--extra-index-url https://download.pytorch.org/whl/cu121

# numpy MUST stay <2: cv_bridge is compiled against numpy 1.x and
# crashes with "_ARRAY_API not found" under numpy 2.x.
numpy==1.26.4

# Deep learning (CUDA 12.1 build). torch==2.12.0 has no published +cu121
# wheel yet — 2.5.1 is the latest release with one available. Bump this
# once a newer +cu121 wheel is published upstream.
torch==2.5.1+cu121

# Vision. The -headless build avoids GUI dependencies (no display in Codespace).
opencv-python-headless==4.10.0.84

# Plotting (trajectory visualization).
matplotlib==3.10.9

# Testing. Recent pytest required so the anyio plugin loads correctly.
pytest==9.1.0
